// Unit tests for authentication criteria.

import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import static org.junit.jupiter.api.Assertions.*;

public class LoginTest {

private Login testUser;

@BeforeEach
public void setUp() {
testUser = new Login("den_1", "38Baby!#", "Denau", "Hermanus", "+27769856209");
}

@Test
public void testUsernameIncorrectFormatMessage() {
testUser.setUsername("denau!!!!!!");
String expected = "Username is not correctly formatted; please ensure that your username contains an underscore and is no more than five characters in length.";
assertEquals(expected, testUser.registerUser());
}

@Test
public void testPasswordIncorrectFormatMessage() {
testUser.setPassword("password");
String expected = "Password is not correctly formatted; please ensure that the password contains at least eight characters, a capital letter, a number, and a special character.";
assertEquals(expected, testUser.registerUser());
}

@Test
public void testCellPhoneIncorrectFormatMessage() {
testUser.setCellPhoneNumber("0769856209");
String expected = "Cell number is incorrectly formatted or does not contain an international code; please correct the number and try again.";
assertEquals(expected, testUser.registerUser());
}

@Test
public void testReturnLoginStatusSuccessMessage() {
String expected = "Welcome Denau, Hermanus it is great to see you again.";
assertEquals(expected, testUser.returnLoginStatus(true));
}

@Test
public void testReturnLoginStatusFailureMessage() {
String expected = "Username or password incorrect, please try again.";
assertEquals(expected, testUser.returnLoginStatus(false));
}

@Test
public void testUsernameCorrectlyFormatted() {
testUser.setUsername("den_1");
assertTrue(testUser.checkUserName());
}

@Test
public void testUsernameIncorrectlyFormatted() {
testUser.setUsername("denau!!!!!!");
assertFalse(testUser.checkUserName());
}

@Test
public void testPasswordMeetsComplexity() {
testUser.setPassword("38Baby!#");
assertTrue(testUser.checkPasswordComplexity());
}

@Test
public void testPasswordDoesNotMeetComplexity() {
testUser.setPassword("password");
assertFalse(testUser.checkPasswordComplexity());
}

@Test
public void testCellPhoneNumberCorrectlyFormatted() {
testUser.setCellPhoneNumber("+27769856209");
assertTrue(testUser.checkCellPhoneNumber());
}

@Test
public void testCellPhoneNumberIncorrectlyFormatted() {
testUser.setCellPhoneNumber("0769856209");
assertFalse(testUser.checkCellPhoneNumber());
}

@Test
public void testLoginSuccessful() {
assertTrue(testUser.loginUser("den_1", "38Baby!#"));
}

@Test
public void testLoginFailed() {
assertFalse(testUser.loginUser("den_1", "wrongPassword"));
}
}
